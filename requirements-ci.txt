# Test-only dependencies for .github/workflows/tests.yml, installed on top of
# requirements.txt.  Product dependencies stay in requirements.txt.

# (.test_durations); pytest-timeout keeps one hung test from burning a whole shard.

pytest-timeout>=2.4
pytest-asyncio>=1.4                     # tests use @pytest.mark.asyncio

# 4.56 the suite was developed on, and qdrant-client must stay within one minor of the
# v1.18 server the tests talk to.
transformers<5
qdrant-client<1.19

# assumption, not a product break, so the pin stays here and not in requirements.txt.
# (starlette<1.4 and mcp<2 are product pins and live in requirements.txt.)
